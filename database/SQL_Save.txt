
CREATE TABLE usuarios (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    senha_hash VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'user',
    data_criacao TIMESTAMP DEFAULT NOW()
);

CREATE TABLE campos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    tipo VARCHAR(50) NOT NULL,
    obrigatorio BOOLEAN NOT NULL DEFAULT false,
    valor_padrao VARCHAR(255)
);

CREATE TABLE modelos_ficha (
    id SERIAL PRIMARY KEY,
    sistema VARCHAR(100) NOT NULL,
    nome VARCHAR(100) NOT NULL,
    descricao TEXT,
    data_criacao TIMESTAMP DEFAULT NOW(),
    criador_id INT NOT NULL,
    
    CONSTRAINT fk_modelo_criador
        FOREIGN KEY (criador_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
);

CREATE TABLE modelos_campos (
    modelo_id INT NOT NULL,
    campo_id INT NOT NULL,

    PRIMARY KEY (modelo_id, campo_id),

    CONSTRAINT fk_mc_modelo
        FOREIGN KEY (modelo_id)
        REFERENCES modelos_ficha(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_mc_campo
        FOREIGN KEY (campo_id)
        REFERENCES campos(id)
        ON DELETE CASCADE
);

CREATE TABLE ficha (
    id SERIAL PRIMARY KEY,
    modelo_id INT NOT NULL,
    usuario_id INT NOT NULL,
    nome_personagem VARCHAR(100) NOT NULL,
    data_criacao TIMESTAMP DEFAULT NOW(),

    CONSTRAINT fk_ficha_modelo
        FOREIGN KEY (modelo_id)
        REFERENCES modelos_ficha(id)
        ON DELETE RESTRICT,

    CONSTRAINT fk_ficha_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
);

CREATE TABLE valores (
    id SERIAL PRIMARY KEY,
    id_campo INT NOT NULL,
    id_ficha INT NOT NULL,
    valor VARCHAR(255),

    CONSTRAINT fk_valor_campo
        FOREIGN KEY (id_campo)
        REFERENCES campos(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_valor_ficha
        FOREIGN KEY (id_ficha)
        REFERENCES ficha(id)
        ON DELETE CASCADE,

    CONSTRAINT unique_valor_por_campo
        UNIQUE (id_campo, id_ficha)
);
