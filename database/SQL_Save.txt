
CREATE TABLE usuarios (
    usr_id SERIAL PRIMARY KEY,
    usr_nome VARCHAR(100) NOT NULL,
    usr_email VARCHAR(150) NOT NULL UNIQUE,
    usr_senha_hash VARCHAR(255) NOT NULL,
    usr_role VARCHAR(50) NOT NULL DEFAULT 'user',
);

CREATE TABLE campos (
    cmp_id SERIAL PRIMARY KEY,
    cmp_nome VARCHAR(100) NOT NULL,
    cmp_tipo VARCHAR(50) NOT NULL,
    cmp_obrigatorio BOOLEAN NOT NULL DEFAULT false,
    cmp_valor_padrao VARCHAR(255)
);

CREATE TABLE modelos_ficha (
    mdf_id SERIAL PRIMARY KEY,
    mdf_sistema VARCHAR(100) NOT NULL,
    mdf_nome VARCHAR(100) NOT NULL,
    mdf_descricao TEXT,
    mdf_criador_id INT NOT NULL,
    
    CONSTRAINT fk_modelo_criador
        FOREIGN KEY (mdf_criador_id)
        REFERENCES usuarios(usr_id)
        ON DELETE CASCADE
);

CREATE TABLE modelos_campos (
    mdc_modelo_id INT NOT NULL,
    mdc_campo_id INT NOT NULL,

    PRIMARY KEY (mdc_modelo_id, mdc_campo_id),

    CONSTRAINT fk_mc_modelo
        FOREIGN KEY (mdc_modelo_id)
        REFERENCES modelos_ficha(mdf_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_mc_campo
        FOREIGN KEY (mdc_campo_id)
        REFERENCES campos(cmp_id)
        ON DELETE CASCADE
);

CREATE TABLE ficha (
    fch_id SERIAL PRIMARY KEY,
    fch_modelo_id INT NOT NULL,
    fch_usuario_id INT NOT NULL,
    fch_nome_personagem VARCHAR(100) NOT NULL,
    
    CONSTRAINT fk_ficha_modelo
        FOREIGN KEY (fch_modelo_id)
        REFERENCES modelos_ficha(mdf_id)
        ON DELETE RESTRICT,

    CONSTRAINT fk_ficha_usuario
        FOREIGN KEY (fch_usuario_id)
        REFERENCES usuarios(usr_id)
        ON DELETE CASCADE
);

CREATE TABLE valores (
    val_id SERIAL PRIMARY KEY,
    val_id_campo INT NOT NULL,
    val_id_ficha INT NOT NULL,
    val_valor VARCHAR(255),

    CONSTRAINT fk_valor_campo
        FOREIGN KEY (val_id_campo)
        REFERENCES campos(cmp_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_valor_ficha
        FOREIGN KEY (val_id_ficha)
        REFERENCES ficha(fch_id)
        ON DELETE CASCADE,

    CONSTRAINT unique_valor_por_campo
        UNIQUE (val_id_campo, val_id_ficha)
);
